cmake_minimum_required(VERSION 3.22.1)
project(topenapi VERSION 0.2.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
option(BUILD_SHARED_LIBS "Build the loadable Tcl extension" ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_path(CJSON_INCLUDE_DIR NAMES cJSON.h PATH_SUFFIXES cjson REQUIRED)
find_library(CJSON_LIBRARY NAMES cjson REQUIRED)
find_path(TCL_INCLUDE_PATH NAMES tcl.h HINTS ${TCL_INCLUDE_DIR}
    PATH_SUFFIXES tcl9.1 tcl9.0 tcl8.6 tcl REQUIRED)
file(STRINGS "${TCL_INCLUDE_PATH}/tcl.h" tcl_version_line REGEX "^#[ \t]*define[ \t]+TCL_VERSION[ \t]+\"")
string(REGEX MATCH "[0-9]+\\.[0-9]+" TOPENAPI_TCL_VERSION "${tcl_version_line}")
get_filename_component(tcl_prefix "${TCL_INCLUDE_PATH}" DIRECTORY)
if(TOPENAPI_TCL_VERSION VERSION_LESS 8.6 OR NOT TOPENAPI_TCL_VERSION VERSION_LESS 10)
    message(FATAL_ERROR "Tcl 8.6 or 9.x headers are required")
endif()
if(TOPENAPI_TCL_VERSION VERSION_LESS 9)
    set(TOPENAPI_TCL_REQUIREMENT "8.6-9")
    set(tcl_stub_names tclstub8.6 tclstub86)
else()
    set(TOPENAPI_TCL_REQUIREMENT "9.0-10")
    set(tcl_stub_names tclstub tclstub${TOPENAPI_TCL_VERSION})
endif()
find_program(TCL_TCLSH NAMES tclsh${TOPENAPI_TCL_VERSION}
    HINTS "${tcl_prefix}/bin")

# The validator can be linked from C/C++ applications independently of Tcl.
add_library(topenapi_core STATIC src/openapi_request_validator.cpp src/topenapi.cpp)
set_target_properties(topenapi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(topenapi_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/topenapi>
    ${CJSON_INCLUDE_DIR})
target_link_libraries(topenapi_core PUBLIC ${CJSON_LIBRARY})

add_library(topenapi src/topenapi_tcl.c src/twebserver_adapter.c src/bind.c)
set_target_properties(topenapi PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(topenapi PUBLIC ${TCL_INCLUDE_PATH})
target_link_libraries(topenapi PUBLIC topenapi_core)
target_compile_definitions(topenapi PRIVATE TOPENAPI_VERSION="${PROJECT_VERSION}")
if(BUILD_SHARED_LIBS)
    find_library(TCL_STUB_LIBRARY NAMES ${tcl_stub_names}
        HINTS ${TCL_LIBRARY_DIR} "${tcl_prefix}/lib" REQUIRED)
    target_compile_definitions(topenapi PRIVATE USE_TCL_STUBS)
    target_link_libraries(topenapi PRIVATE ${TCL_STUB_LIBRARY})
    set(TOPENAPI_LOAD "[list load [file join $dir ${CMAKE_SHARED_LIBRARY_PREFIX}topenapi${CMAKE_SHARED_LIBRARY_SUFFIX}] Topenapi]")
else()
    target_compile_definitions(topenapi PRIVATE STATIC_BUILD)
    set(TOPENAPI_LOAD "[list load {} Topenapi]")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(topenapi PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(topenapi_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
set(TOPENAPI_PACKAGE_DIR "${CMAKE_INSTALL_LIBDIR}/topenapi${PROJECT_VERSION}")
install(TARGETS topenapi topenapi_core
    LIBRARY DESTINATION "${TOPENAPI_PACKAGE_DIR}"
    RUNTIME DESTINATION "${TOPENAPI_PACKAGE_DIR}"
    ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl" DESTINATION "${TOPENAPI_PACKAGE_DIR}")
install(FILES src/topenapi.h src/topenapi_tcl.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/topenapi")

if(BUILD_TESTING)
    add_executable(test_validator tests/validator.cpp)
    target_link_libraries(test_validator PRIVATE topenapi_core)
    add_test(NAME validator COMMAND test_validator)
    add_executable(test_c_api tests/c_api.c)
    target_link_libraries(test_c_api PRIVATE topenapi_core)
    add_test(NAME c_api COMMAND test_c_api)
    if(BUILD_SHARED_LIBS)
        if(NOT TCL_TCLSH)
            message(FATAL_ERROR "TCL_TCLSH is required to test shared builds")
        endif()
        set(test_shell "${TCL_TCLSH}")
    else()
        find_library(TCL_LIBRARY NAMES tcl${TOPENAPI_TCL_VERSION}
            HINTS "${tcl_prefix}/lib" REQUIRED)
        add_executable(test_embed tests/embed.c)
        target_link_libraries(test_embed PRIVATE topenapi ${TCL_LIBRARY})
        set(test_shell "$<TARGET_FILE:test_embed>")
    endif()
    add_test(NAME tcl COMMAND ${test_shell} "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl")
    add_test(NAME bind_example COMMAND ${test_shell} "${CMAKE_CURRENT_SOURCE_DIR}/examples/bind.tcl")
    set_tests_properties(tcl bind_example PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    configure_file(tests/install.cmake.in tests/install.cmake @ONLY)
    add_test(NAME install_layout COMMAND ${CMAKE_COMMAND} -DCONFIG=$<CONFIG>
        -P "${CMAKE_CURRENT_BINARY_DIR}/tests/install.cmake")
endif()
